```typescript
import { json } from '@sveltejs/kit';
import type { RequestHandler } from './$types';
import { listMembers, type MemberInput } from '#lib/server/memberStore';
import { validateMemberInput } from '#lib/utils/memberValidation';

export const GET: RequestHandler = async ({ url }) => {
  const page = parseInt(url.searchParams.get('page') || '1');
  const perPage = parseInt(url.searchParams.get('perPage') || '20');
  const limit = Math.min(Math.max(perPage, 1), 100); // Max 100 per page
  const offset = (page - 1) * limit;

  const members = listMembers().slice(offset, offset + limit);
  const total = listMembers().length;
  const totalPages = Math.ceil(total / limit);

  return json({
    members,
    pagination: {
      page,
      perPage: limit,
      total,
      totalPages
    }
  });
};

export const POST: RequestHandler = async ({ request }) => {
  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return json({ errors: { body: 'Invalid JSON' } }, { status: 400 });
  }

  const input = (body && typeof body === 'object' ? body : {}) as Partial<
    Record<keyof MemberInput, unknown>
  >;
  const errors = validateMemberInput(input);
  if (Object.keys(errors).length > 0) return json({ errors }, { status: 400 });

  return json(createMember(input as MemberInput), { status: 201 });
};
```

```typescript
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(100, '1 m'), // 100 requests per minute
  analytics: true
});

export async function checkRateLimit(identifier: string) {
  const { success, limit, remaining, reset } = await ratelimit.limit(identifier);
  return {
    success,
    limit,
    remaining,
    reset
  };
}
```

```typescript
import { json } from '@sveltejs/kit';
import type { RequestHandler } from './$types';
import { listMembers, createMember, type MemberInput } from '#lib/server/memberStore';
import { validateMemberInput } from '#lib/utils/memberValidation';
import { checkRateLimit } from '#lib/server/rateLimit';

export const GET: RequestHandler = async ({ url, request }) => {
  const clientIp = request.headers.get('x-forwarded-for') || 'unknown';
  const { success } = await checkRateLimit(clientIp);

  if (!success) {
    return json({ error: 'Rate limit exceeded' }, { status: 429 });
  }

  const corsHeaders = {
    'Access-Control-Allow-Origin': process.env.FRONTEND_URL || 'http://localhost:5173',
    'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type, Authorization'
  };

  // Handle preflight requests
  if (request.method === 'OPTIONS') {
    return new Response(null, {
      status: 204,
      headers: corsHeaders
    });
  }

  const page = parseInt(url.searchParams.get('page') || '1');
  const perPage = parseInt(url.searchParams.get('perPage') || '20');

  const total = listMembers().length;

  return json(
    {
      members,
      pagination: {
        page,
        perPage: limit,
        total,
        totalPages
      }
    },
    { headers: corsHeaders }
  );
};
```

```typescript
import { error, json } from '@sveltejs/kit';
import type { RequestHandler } from './$types';
import { findMember, updateMember, type MemberInput } from '#lib/server/memberStore';
import { validateMemberInput } from '#lib/utils/memberValidation';

export const GET: RequestHandler = ({ params }) => {
  const member = findMember(params.id);
  if (!member) error(404, 'Member not found');
  return json(member);
};

export const PATCH: RequestHandler = async ({ params, request }) => {
  const existing = findMember(params.id);
  if (!existing) error(404, 'Member not found');

  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return json({ errors: { body: 'Invalid JSON' } }, { status: 400 });
  }

  const patch = (body && typeof body === 'object' ? body : {}) as Partial<MemberInput>;
  const merged: MemberInput = {
    nama: patch.nama ?? existing.nama,
    email: patch.email ?? existing.email,
    hp: patch.hp ?? existing.hp,
    status: patch.status ?? existing.status
  };

  const errors = validateMemberInput(merged);
  if (Object.keys(errors).length > 0) return json({ errors }, { status: 400 });

  return json(updateMember(params.id, merged));
};
```
